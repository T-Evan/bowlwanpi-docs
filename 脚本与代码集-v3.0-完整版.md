### 1. MCP服务器 (mcp_server.py)

**位置**: `~/.openclaw/workspace/mcp_server.py`

**功能**: 提供系统状态、记忆管理、定时任务查询等MCP工具

**代码**:
```python
#!/usr/bin/env python3
"""
BowlWanpi MCP服务器
提供记忆存储/检索、系统状态查询等功能
"""

import asyncio
import json
import os
import sys
from datetime import datetime
from typing import Any, Dict, List, Optional

# MCP协议导入
try:
    from mcp.server import Server
    from mcp.types import TextContent, Tool
except ImportError:
    print("错误: 缺少mcp依赖，请安装: pip install mcp", file=sys.stderr)
    sys.exit(1)

# 技能导入
try:
    from skills.unified_memory.unified_memory_manager import (
        store_to_all_systems,
        retrieve_merged
    )
except ImportError:
    store_to_all_systems = None
    retrieve_merged = None

# 创建MCP服务器
server = Server("bowlwanpi")

@server.list_tools()
async def list_tools() -> List[Tool]:
    """列出所有可用工具"""
    return [
        Tool(
            name="get_system_status",
            description="获取系统状态信息(CPU/内存/磁盘/网络)",
            inputSchema={
                "type": "object",
                "properties": {},
                "required": []
            }
        ),
        Tool(
            name="search_memory",
            description="从记忆系统中检索相关信息",
            inputSchema={
                "type": "object",
                "properties": {
                    "query": {
                        "type": "string",
                        "description": "搜索关键词"
                    },
                    "limit": {
                        "type": "integer",
                        "description": "返回结果数量",
                        "default": 5
                    }
                },
                "required": ["query"]
            }
        ),
        Tool(
            name="remember_fact",
            description="存储重要信息到记忆系统",
            inputSchema={
                "type": "object",
                "properties": {
                    "text": {
                        "type": "string",
                        "description": "要存储的信息"
                    },
                    "category": {
                        "type": "string",
                        "description": "信息类别",
                        "default": "general"
                    },
                    "importance": {
                        "type": "number",
                        "description": "重要性评分(0-1)",
                        "default": 0.5
                    }
                },
                "required": ["text"]
            }
        ),
        Tool(
            name="get_daily_schedule",
            description="获取今日定时任务列表",
            inputSchema={
                "type": "object",
                "properties": {},
                "required": []
            }
        ),
        Tool(
            name="get_moltbook_info",
            description="获取Moltbook账号信息",
            inputSchema={
                "type": "object",
                "properties": {},
                "required": []
            }
        )
    ]

@server.call_tool()
async def call_tool(name: str, arguments: Dict[str, Any]) -> List[TextContent]:
    """调用工具"""
    
    if name == "get_system_status":
        return await handle_get_system_status(arguments)
    elif name == "search_memory":
        return await handle_search_memory(arguments)
    elif name == "remember_fact":
        return await handle_remember_fact(arguments)
    elif name == "get_daily_schedule":
        return await handle_get_daily_schedule(arguments)
    elif name == "get_moltbook_info":
        return await handle_get_moltbook_info(arguments)
    else:
        return [TextContent(type="text", text=f"未知工具: {name}")]

async def handle_get_system_status(arguments: Dict[str, Any]) -> List[TextContent]:
    """获取系统状态"""
    try:
        import psutil
        
        # CPU信息
        cpu_percent = psutil.cpu_percent(interval=1)
        cpu_count = psutil.cpu_count()
        
        # 内存信息
        memory = psutil.virtual_memory()
        
        # 磁盘信息
        disk = psutil.disk_usage('/')
        
        # 网络信息
        net_io = psutil.net_io_counters()
        
        status = {
            "timestamp": datetime.now().isoformat(),
            "cpu": {
                "percent": cpu_percent,
                "count": cpu_count
            },
            "memory": {
                "total_gb": round(memory.total / (1024**3), 2),
                "used_gb": round(memory.used / (1024**3), 2),
                "percent": memory.percent
            },
            "disk": {
                "total_gb": round(disk.total / (1024**3), 2),
                "used_gb": round(disk.used / (1024**3), 2),
                "percent": round(disk.used / disk.total * 100, 2)
            },
            "network": {
                "bytes_sent_mb": round(net_io.bytes_sent / (1024**2), 2),
                "bytes_recv_mb": round(net_io.bytes_recv / (1024**2), 2)
            }
        }
        
        return [TextContent(type="text", text=json.dumps(status, indent=2, ensure_ascii=False))]
    except Exception as e:
        return [TextContent(type="text", text=f"获取系统状态失败: {str(e)}")]

async def handle_search_memory(arguments: Dict[str, Any]) -> List[TextContent]:
    """搜索记忆"""
    if retrieve_merged is None:
        return [TextContent(type="text", text="记忆系统未安装")]
    
    try:
        query = arguments.get("query", "")
        limit = arguments.get("limit", 5)
        
        memories = await retrieve_merged(query, limit=limit)
        
        if not memories:
            return [TextContent(type="text", text="未找到相关记忆")]
        
        result = []
        for i, mem in enumerate(memories, 1):
            result.append(f"[{i}] {mem.get('content', 'N/A')}")
            result.append(f"    来源: {mem.get('source', 'unknown')}")
            result.append(f"    时间: {mem.get('timestamp', 'N/A')}")
            result.append("")
        
        return [TextContent(type="text", text="\n".join(result))]
    except Exception as e:
        return [TextContent(type="text", text=f"搜索记忆失败: {str(e)}")]

async def handle_remember_fact(arguments: Dict[str, Any]) -> List[TextContent]:
    """存储记忆"""
    if store_to_all_systems is None:
        return [TextContent(type="text", text="记忆系统未安装")]
    
    try:
        text = arguments.get("text", "")
        category = arguments.get("category", "general")
        importance = arguments.get("importance", 0.5)
        
        await store_to_all_systems(
            user_message="",
            assistant_response=text,
            metadata={
                "category": category,
                "importance": importance
            }
        )
        
        return [TextContent(type="text", text=f"✅ 已存储到记忆系统 (类别: {category}, 重要性: {importance})")]
    except Exception as e:
        return [TextContent(type="text", text=f"存储记忆失败: {str(e)}")]

async def handle_get_daily_schedule(arguments: Dict[str, Any]) -> List[TextContent]:
    """获取定时任务"""
    try:
        # 读取cron配置
        cron_file = "/etc/cron.d/bowlwanpi-backup-cron"
        if not os.path.exists(cron_file):
            return [TextContent(type="text", text="未找到定时任务配置")]
        
        with open(cron_file, 'r') as f:
            lines = f.readlines()
        
        tasks = []
        for line in lines:
            line = line.strip()
            if line and not line.startswith('#') and line.startswith('*'):
                parts = line.split()
                if len(parts) >= 7:
                    time = ' '.join(parts[:5])
                    command = ' '.join(parts[6:])
                    tasks.append(f"{time}: {command}")
        
        return [TextContent(type="text", text="定时任务列表:\n" + "\n".join(tasks))]
    except Exception as e:
        return [TextContent(type="text", text=f"获取定时任务失败: {str(e)}")]

async def handle_get_moltbook_info(arguments: Dict[str, Any]) -> List[TextContent]:
    """获取Moltbook信息"""
    try:
        import json
        creds_file = "/root/.openclaw/workspace/secrets/moltbook-credentials.json"
        
        if not os.path.exists(creds_file):
            return [TextContent(type="text", text="未找到Moltbook凭证")]
        
        with open(creds_file, 'r') as f:
            creds = json.load(f)
        
        info = {
            "agent_name": "BowlWanpi",
            "agent_id": creds.get("agent_id", "N/A"),
            "profile_url": f"https://moltbook.com/u/BowlWanpi",
            "status": "已配置"
        }
        
        return [TextContent(type="text", text=json.dumps(info, indent=2, ensure_ascii=False))]
    except Exception as e:
        return [TextContent(type="text", text=f"获取Moltbook信息失败: {str(e)}")]

# 主函数
async def main():
    """启动MCP服务器"""
    from mcp.server.stdio import stdio_server
    
    async with stdio_server() as (read_stream, write_stream):
        await server.run(
            read_stream,
            write_stream,
            server.create_initialization_options()
        )

if __name__ == "__main__":
    asyncio.run(main())
```

**使用方式**:
```bash

python3 mcp_server.py

# 或作为MCP服务
# 配置在 mcp_config.json 中
```

### 3. 心跳脚本 (heartbeat.sh)

**位置**: `~/.openclaw/workspace/scripts/heartbeat.sh`

**功能**: 每5分钟检查系统状态并生成报告

**代码**:
```bash
#!/bin/bash
# BowlWanpi 心跳脚本 v3.0
# 执行频率: 每5分钟

# 配置
WORKSPACE="/root/.openclaw/workspace"
LOG_DIR="/var/log"
LOG_FILE="${LOG_DIR}/bowlwanpi-heartbeat.log"
REPORT_FILE="${LOG_DIR}/bowlwanpi-latest-report.txt"
ALERT_COOLDOWN=3600  # 告警冷却时间(秒)

# 日志函数
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> "$LOG_FILE"
}

# 检查进程
check_process() {
    local name=$1
    local pattern=$2
    
    if pgrep -f "$pattern" > /dev/null; then
        echo "running"
    else
        echo "stopped"
    fi
}

# 检查端口
check_port() {
    local port=$1
    
    if netstat -tlnp 2>/dev/null | grep -q ":$port "; then
        echo "open"
    else
        echo "closed"
    fi
}

# 主检查逻辑
main() {
    local timestamp=$(date '+%Y-%m-%d %H:%M:%S')
    local alerts=()
    
    # 1. 检查Mihomo代理
    local mihomo_status=$(check_process "mihomo" "mihomo")
    local proxy_port=$(check_port 7890)
    
    # 2. 检查OpenClaw网关
    local openclaw_port=$(check_port 18789)
    
    # 3. 检查系统资源
    local cpu_usage=$(top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'%' -f1)
    local mem_usage=$(free | grep Mem | awk '{printf "%.0f", $3/$2 * 100.0}')
    local disk_usage=$(df / | tail -1 | awk '{print $5}' | sed 's/%//')
    
    # 生成报告
    cat > "$REPORT_FILE" << EOF
BowlWanpi 系统状态报告
生成时间: $timestamp

服务状态:
---------
Mihomo代理: $mihomo_status
代理端口(7890): $proxy_port
OpenClaw端口(18789): $openclaw_port

系统资源:
---------
CPU使用率: ${cpu_usage}%
内存使用率: ${mem_usage}%
磁盘使用率: ${disk_usage}%

检查项:
-------
EOF

    # 检查告警条件
    if [ "$mihomo_status" = "stopped" ]; then
        echo "❌ Mihomo代理未运行" >> "$REPORT_FILE"
        alerts+=("Mihomo代理已停止")
    else
        echo "✅ Mihomo代理运行正常" >> "$REPORT_FILE"
    fi
    
    if [ "$proxy_port" = "closed" ]; then
        echo "❌ 代理端口(7890)未开放" >> "$REPORT_FILE"
        alerts+=("代理端口未开放")
    else
        echo "✅ 代理端口正常" >> "$REPORT_FILE"
    fi
    
    if [ "$openclaw_port" = "closed" ]; then
        echo "⚠️  OpenClaw端口(18789)未开放" >> "$REPORT_FILE"
        # 不告警，可能是有意停止
    else
        echo "✅ OpenClaw端口正常" >> "$REPORT_FILE"
    fi
    
    if [ "${disk_usage%.*}" -gt 90 ]; then
        echo "❌ 磁盘使用率过高: ${disk_usage}%" >> "$REPORT_FILE"
        alerts+=("磁盘使用率超过90%")
    elif [ "${disk_usage%.*}" -gt 80 ]; then
        echo "⚠️  磁盘使用率偏高: ${disk_usage}%" >> "$REPORT_FILE"
    else
        echo "✅ 磁盘使用率正常: ${disk_usage}%" >> "$REPORT_FILE"
    fi
    
    # 记录日志
    log "心跳检查完成 - CPU:${cpu_usage}% MEM:${mem_usage}% DISK:${disk_usage}%"
    
    # 发送告警 (如果满足条件)
    if [ ${#alerts[@]} -gt 0 ]; then
        local alert_key="/tmp/bowlwanpi-alert-last"
        local current_time=$(date +%s)
        local last_alert=0
        
        if [ -f "$alert_key" ]; then
            last_alert=$(cat "$alert_key")
        fi
        
        local time_diff=$((current_time - last_alert))
        
        if [ $time_diff -gt $ALERT_COOLDOWN ]; then
            # 发送告警通知
            local alert_msg="🚨 BowlWanpi 告警:\n"
            for alert in "${alerts[@]}"; do
                alert_msg="${alert_msg}- ${alert}\n"
            done
            
            # 发送到Feishu (如果配置了)
            # echo -e "$alert_msg" | openclaw message send --channel feishu --target "your-id" --message -
            
            echo "$current_time" > "$alert_key"
            log "告警已发送: ${alerts[*]}"
        fi
    fi
}

# 执行主函数
main "$@"

exit 0
```

**使用方式**:
```bash

# 查看最新报告

# 查看历史日志
tail -f /var/log/bowlwanpi-heartbeat.log
```
